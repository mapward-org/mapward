
## Дополнения решений

1. include exclude глобы для object-children-map

2. Адрес без решётки резолвится в абсолютный файловый путь: `mapward://` — корень карты,
`mapward://~` — папка текущего объекта. Иначе непонятно, от чего считать относительные
пути в конфигах, а cwd мы не контролируем

3. Подстановка происходит после extends всех. Поэтому ~ в путях значит конкретный обьект а не прототип

## 1. Добавление базовой карты
На основе решений `/ru/docs/decisions/(0002,0003,0004,0005,0006)`
Давай сделаем несколько базовых объектов на карте ru/map 


Текст директив и экшонов - придумай сам минимальную базу


map-state.json - не создаётся руками. Он появится при первом рендере карты будущим расширением



### Что на карте


```yaml
map/:
  _index.json:                  # корневой объект карты
                                # layout: списки prototypes и shared-metrics строкой над картой,
                                # areas [[prototypes, shared-metrics], [children, children]]
                                # в props.basePath - абсолютный путь до корня репо: ${{ mapward:// }}/../..
  _metrics/:
    prototypes/:                # список прототипов ссылками, label "Прототипы"
      config.json:              # collect: script, однострочник на node отдаёт {items},
                                # display: list. object-children-map тут не годится —
                                # он отдаёт форму для карты, а списку нужны items
                                # прототипы объекты, ссылка ведёт на их страницу mapward://
    shared-metrics/:            # список общих метрик ссылками, label "Общие метрики"
      config.json:              # collect: script, однострочник на node отдаёт {items} с путями
                                # к config.json каждой метрики, display: list
                                # метрики не объекты, ссылка открывает config.json
    children/:                  # дети карты картой (исключая prototypes и shared-metrics)
      config.json:              # collect: object-children-map, display: map

  shared-metrics/:              # группа, своего _index.json нет
    files/:
      config.json:              # collect: read-dir basePath "${{ mapward://#props.basePath }}/${{ mapward://~#props.codePath }}", display: tree

  prototypes/:                  # группа, своего _index.json нет
    package/:
      _index.json:              # name: "Пакет". Так же есть пропс fullPath который склеивает значение из корня карты + codePath который задаётся в наследнике
      _metrics/:
        code/:                  # static: link "${{ mapward://~#props.codePath }}", display: link
        docs/:                  # static: link "${{ mapward://~#props.docsPath }}", display: link
        version/:               # collect: script, читает версию из package.json по fullPath
        tests/:                 # collect: script, прогон тестов, display: status
        lint/:                  # collect: script, прогон линтера, display: status
        public-api/:            # collect: prompt, собирает public api в collect.json, display: link 
        files/:                 # extends mapward://shared-metrics/files 
      _directives/:
        create-package.md:      # как создать пустой пакет. Имя смысловое — её наследуют. Просит в props указать путь к пакету и путь к доке
      _actions/:
        publish-version.md:     # как правильно опубликовать версию

  packages/:
    core/:
      _index.json:                # extends mapward://prototypes/package
                                  # props: codePath packages/core, docsPath ru/docs/core.md, packageName
                                  # своих метрик нет вообще — прототип работает на props
  apps/:
    vscode-application/:
      _index.json:                # props: codePath apps/vscode-extension
      _metrics/:
        code/:                    # static: link "${{ mapward://~#props.codePath }}", display: link
        files/:                   # extends mapward://shared-metrics/files
      _directives/:
        create-base.md:           # создать основу расширения. Не запускаем сейчас
      _actions/:
        run-dev.md:               # запустить dev версию

  relations/:                   # группа, своего _index.json нет
    vscode-application-to-core/:
      _index.json:              # props: from mapward://apps/vscode-application, to mapward://packages/core
      _metrics/:
        code/:                  # дальше будет собирать импорты и давать на них ссылки
```
